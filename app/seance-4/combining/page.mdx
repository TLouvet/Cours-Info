# Combiner useEffect & Routing

Le vrai pouvoir vient de la combinaison de `useEffect` et du routing pour créer des applications multi-pages avec données dynamiques.

## Pattern Principal : Fetch Basé sur les Routes

Le pattern le plus courant : récupérer des données différentes selon l'URL.

```javascript
import { useParams } from 'react-router-dom';
import { useState, useEffect } from 'react';

function UserProfile() {
  const { userId } = useParams(); // Récupérer l'ID depuis l'URL
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchUser() {
      setLoading(true);

      const response = await fetch(
        `https://jsonplaceholder.typicode.com/users/${userId}`
      );
      const data = await response.json();

      setUser(data);
      setLoading(false);
    }

    fetchUser();
  }, [userId]); // 👈 Re-fetch quand userId change

  if (loading) return <div>Chargement...</div>;

  return (
    <div>
      <h1>{user.name}</h1>
      <p>{user.email}</p>
    </div>
  );
}

// Route
<Route path="/users/:userId" element={<UserProfile />} />
```

**Timeline :**
```
1. User navigue vers /users/1
   → useParams() retourne { userId: "1" }
   → useEffect se déclenche
   → Fetch user 1
   → Affiche user 1

2. User navigue vers /users/2
   → useParams() retourne { userId: "2" }
   → userId a changé → useEffect se re-déclenche
   → Fetch user 2
   → Affiche user 2
```

<Callout type="tip" title="Dépendance critique">
**Toujours mettre les params d'URL dans les dépendances de useEffect !**

```javascript
// ✅ Bon : re-fetch si userId change
useEffect(() => {
  fetch(`/api/users/${userId}`);
}, [userId]);

// ❌ Mauvais : fetch une seule fois, ignore les changements d'URL
useEffect(() => {
  fetch(`/api/users/${userId}`);
}, []);
```
</Callout>

## Exemple Complet : Annuaire Utilisateurs

```javascript
import { Routes, Route, Link, useParams } from 'react-router-dom';
import { useState, useEffect } from 'react';

// Page 1 : Liste des utilisateurs
function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchUsers() {
      const response = await fetch('https://jsonplaceholder.typicode.com/users');
      const data = await response.json();
      setUsers(data);
      setLoading(false);
    }

    fetchUsers();
  }, []);

  if (loading) return <div>Chargement...</div>;

  return (
    <div>
      <h1>Utilisateurs</h1>
      <ul>
        {users.map(user => (
          <li key={user.id}>
            <Link to={`/users/${user.id}`}>
              {user.name}
            </Link>
          </li>
        ))}
      </ul>
    </div>
  );
}

// Page 2 : Profil utilisateur avec ses posts
function UserProfile() {
  const { userId } = useParams();
  const [user, setUser] = useState(null);
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchUserData() {
      setLoading(true);

      // Fetch en parallèle
      const [userResponse, postsResponse] = await Promise.all([
        fetch(`https://jsonplaceholder.typicode.com/users/${userId}`),
        fetch(`https://jsonplaceholder.typicode.com/posts?userId=${userId}`)
      ]);

      const userData = await userResponse.json();
      const postsData = await postsResponse.json();

      setUser(userData);
      setPosts(postsData);
      setLoading(false);
    }

    fetchUserData();
  }, [userId]); // Re-fetch si on navigue vers un autre utilisateur

  if (loading) return <div>Chargement...</div>;

  return (
    <div>
      <Link to="/users">← Retour à la liste</Link>

      <h1>{user.name}</h1>
      <p>Email : {user.email}</p>
      <p>Téléphone : {user.phone}</p>
      <p>Site web : {user.website}</p>

      <h2>Articles de {user.name}</h2>
      <ul>
        {posts.map(post => (
          <li key={post.id}>
            <Link to={`/posts/${post.id}`}>
              {post.title}
            </Link>
          </li>
        ))}
      </ul>
    </div>
  );
}

// App
function App() {
  return (
    <Routes>
      <Route path="/users" element={<UserList />} />
      <Route path="/users/:userId" element={<UserProfile />} />
    </Routes>
  );
}
```

<Callout type="success" title="Promise.all pour Performance">
Au lieu de faire 2 fetch séquentiels :

```javascript
// ❌ Lent : 2 requêtes séquentielles
const userRes = await fetch(`/users/${userId}`);
const user = await userRes.json();

const postsRes = await fetch(`/posts?userId=${userId}`);
const posts = await postsRes.json();

// ✅ Rapide : 2 requêtes en parallèle
const [userRes, postsRes] = await Promise.all([
  fetch(`/users/${userId}`),
  fetch(`/posts?userId=${userId}`)
]);

const user = await userRes.json();
const posts = await postsRes.json();
```

`Promise.all` lance toutes les requêtes en même temps !
</Callout>

## Pattern : Liste → Détail

Le pattern le plus courant en React Router + useEffect.

```javascript
// Routes
<Routes>
  <Route path="/products" element={<ProductList />} />
  <Route path="/products/:id" element={<ProductDetail />} />
</Routes>

// Page Liste
function ProductList() {
  const [products, setProducts] = useState([]);

  useEffect(() => {
    fetch('/api/products')
      .then(res => res.json())
      .then(data => setProducts(data));
  }, []);

  return (
    <ul>
      {products.map(product => (
        <li key={product.id}>
          <Link to={`/products/${product.id}`}>
            {product.name}
          </Link>
        </li>
      ))}
    </ul>
  );
}

// Page Détail
function ProductDetail() {
  const { id } = useParams();
  const [product, setProduct] = useState(null);

  useEffect(() => {
    fetch(`/api/products/${id}`)
      .then(res => res.json())
      .then(data => setProduct(data));
  }, [id]);

  if (!product) return <div>Chargement...</div>;

  return (
    <div>
      <Link to="/products">← Retour</Link>
      <h1>{product.name}</h1>
      <p>{product.description}</p>
      <p>Prix : {product.price}€</p>
    </div>
  );
}
```

## Pattern : Recherche avec Search Params

Utiliser les search params pour des filtres dynamiques.

```javascript
import { useSearchParams } from 'react-router-dom';

function ProductList() {
  const [searchParams, setSearchParams] = useSearchParams();
  const [products, setProducts] = useState([]);

  const category = searchParams.get('category') || 'all';
  const sort = searchParams.get('sort') || 'name';

  useEffect(() => {
    async function fetchProducts() {
      // Fetch avec filtres
      const response = await fetch(
        `/api/products?category=${category}&sort=${sort}`
      );
      const data = await response.json();
      setProducts(data);
    }

    fetchProducts();
  }, [category, sort]); // Re-fetch si les filtres changent

  return (
    <div>
      <h1>Produits</h1>

      {/* Filtres */}
      <div>
        <select
          value={category}
          onChange={(e) => setSearchParams({ category: e.target.value, sort })}
        >
          <option value="all">Toutes</option>
          <option value="electronics">Électronique</option>
          <option value="books">Livres</option>
        </select>

        <select
          value={sort}
          onChange={(e) => setSearchParams({ category, sort: e.target.value })}
        >
          <option value="name">Nom</option>
          <option value="price">Prix</option>
        </select>
      </div>

      {/* Liste */}
      <ul>
        {products.map(product => (
          <li key={product.id}>{product.name}</li>
        ))}
      </ul>
    </div>
  );
}
```

**URLs générées :**
```
/products → Tous, tri par nom
/products?category=books → Livres, tri par nom
/products?category=electronics&sort=price → Électronique, tri par prix
```

## Pattern : Pagination

```javascript
function PostList() {
  const [searchParams, setSearchParams] = useSearchParams();
  const [posts, setPosts] = useState([]);

  const page = parseInt(searchParams.get('page') || '1', 10);
  const limit = 10;

  useEffect(() => {
    async function fetchPosts() {
      const start = (page - 1) * limit;
      const response = await fetch(
        `https://jsonplaceholder.typicode.com/posts?_start=${start}&_limit=${limit}`
      );
      const data = await response.json();
      setPosts(data);
    }

    fetchPosts();
  }, [page]);

  function nextPage() {
    setSearchParams({ page: page + 1 });
  }

  function prevPage() {
    if (page > 1) {
      setSearchParams({ page: page - 1 });
    }
  }

  return (
    <div>
      <h1>Posts - Page {page}</h1>

      <ul>
        {posts.map(post => (
          <li key={post.id}>{post.title}</li>
        ))}
      </ul>

      <div>
        <button onClick={prevPage} disabled={page === 1}>
          ← Précédent
        </button>

        <span>Page {page}</span>

        <button onClick={nextPage}>
          Suivant →
        </button>
      </div>
    </div>
  );
}
```

**URLs :**
```
/posts → Page 1
/posts?page=2 → Page 2
/posts?page=3 → Page 3
```

## Pattern : Redirect Conditionnel

Rediriger basé sur des données fetchées.

```javascript
import { useNavigate } from 'react-router-dom';

function ProtectedPage() {
  const navigate = useNavigate();
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function checkAuth() {
      try {
        const response = await fetch('/api/me');

        if (!response.ok) {
          // Pas connecté → rediriger vers login
          navigate('/login');
          return;
        }

        const data = await response.json();
        setUser(data);

      } catch (error) {
        navigate('/login');
      } finally {
        setLoading(false);
      }
    }

    checkAuth();
  }, [navigate]);

  if (loading) return <div>Vérification...</div>;

  return (
    <div>
      <h1>Bienvenue {user.name}</h1>
    </div>
  );
}
```

## Pattern : Breadcrumbs Dynamiques

Afficher le fil d'Ariane avec données fetchées.

```javascript
function ProductDetail() {
  const { categoryId, productId } = useParams();
  const [category, setCategory] = useState(null);
  const [product, setProduct] = useState(null);

  useEffect(() => {
    async function fetchData() {
      const [catRes, prodRes] = await Promise.all([
        fetch(`/api/categories/${categoryId}`),
        fetch(`/api/products/${productId}`)
      ]);

      setCategory(await catRes.json());
      setProduct(await prodRes.json());
    }

    fetchData();
  }, [categoryId, productId]);

  if (!category || !product) return <div>Chargement...</div>;

  return (
    <div>
      {/* Breadcrumbs avec noms réels */}
      <nav>
        <Link to="/">Accueil</Link> /
        <Link to="/categories">Catégories</Link> /
        <Link to={`/categories/${categoryId}`}>{category.name}</Link> /
        <span>{product.name}</span>
      </nav>

      <h1>{product.name}</h1>
      <p>{product.description}</p>
    </div>
  );
}
```

## Gérer les Erreurs 404

Vérifier si la ressource existe et afficher une 404 sinon.

```javascript
function UserProfile() {
  const { userId } = useParams();
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [notFound, setNotFound] = useState(false);

  useEffect(() => {
    async function fetchUser() {
      try {
        setLoading(true);
        setNotFound(false);

        const response = await fetch(
          `https://jsonplaceholder.typicode.com/users/${userId}`
        );

        if (!response.ok) {
          setNotFound(true);
          return;
        }

        const data = await response.json();

        // API retourne {} pour un user inexistant
        if (!data.id) {
          setNotFound(true);
          return;
        }

        setUser(data);

      } catch (error) {
        setNotFound(true);
      } finally {
        setLoading(false);
      }
    }

    fetchUser();
  }, [userId]);

  if (loading) return <div>Chargement...</div>;

  if (notFound) {
    return (
      <div>
        <h1>Utilisateur non trouvé</h1>
        <Link to="/users">← Retour à la liste</Link>
      </div>
    );
  }

  return (
    <div>
      <h1>{user.name}</h1>
    </div>
  );
}
```

## Cleanup avec AbortController

Annuler les fetch si l'utilisateur change de page rapidement.

```javascript
function ProductDetail() {
  const { productId } = useParams();
  const [product, setProduct] = useState(null);

  useEffect(() => {
    const controller = new AbortController();

    async function fetchProduct() {
      try {
        const response = await fetch(
          `/api/products/${productId}`,
          { signal: controller.signal }
        );

        const data = await response.json();
        setProduct(data);

      } catch (error) {
        if (error.name !== 'AbortError') {
          console.error('Error:', error);
        }
      }
    }

    fetchProduct();

    return () => {
      controller.abort(); // Annuler le fetch si on change de page
    };
  }, [productId]);

  if (!product) return <div>Chargement...</div>;

  return <div>{product.name}</div>;
}
```

<Callout type="warning" title="Pourquoi c'est important ?">
Sans AbortController :

```
User clique produit 1 → Fetch produit 1 (lent)
User clique produit 2 → Fetch produit 2 (rapide)

Réponse produit 2 arrive → Affiche produit 2 ✅
Réponse produit 1 arrive → Affiche produit 1 ❌ Mauvais !
```

L'AbortController annule la requête du produit 1 quand on navigue vers le produit 2.
</Callout>

## Récapitulatif

<Callout type="success" title="Patterns Essentiels">
**1. Fetch basé sur params :**
```javascript
const { id } = useParams();
useEffect(() => {
  fetch(`/api/items/${id}`);
}, [id]);
```

**2. Liste → Détail :**
```javascript
// Liste : fetch tous
// Détail : fetch un seul basé sur l'URL
```

**3. Search params pour filtres :**
```javascript
const [searchParams] = useSearchParams();
const filter = searchParams.get('category');
useEffect(() => {
  fetch(`/api/items?category=${filter}`);
}, [filter]);
```

**4. Promise.all pour parallélisme :**
```javascript
const [data1, data2] = await Promise.all([
  fetch(url1),
  fetch(url2)
]);
```

**5. AbortController pour cleanup :**
```javascript
useEffect(() => {
  const controller = new AbortController();
  fetch(url, { signal: controller.signal });
  return () => controller.abort();
}, [url]);
```
</Callout>
